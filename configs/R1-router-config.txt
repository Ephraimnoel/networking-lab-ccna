! ==============================================================
! NORTHSTAR SERVICES - EDGE ROUTER CONFIGURATION
! Device: Cisco 2911 / 1941 ISR
! Hostname: Northstar-R1
! Purpose: Router-on-a-Stick (ROAS) Inter-VLAN Routing & DHCP Server
! Environment: Cisco Packet Tracer Simulation
! ==============================================================

enable
configure terminal

! --- Basic Device Identity & Security ---
hostname Northstar-R1
no ip domain-lookup
service password-encryption
enable secret LabEnableSecret123!

! --- Security Warning Banner ---
banner motd #
=============================================================
[!] NORTHSTAR SERVICES - RESTRICTED ACCESS ONLY
Unauthorized access or usage is strictly prohibited.
All network activities are logged and monitored.
=============================================================
#

! --- Management Plane Hardening (SSHv2 & Local Auth) ---
ip domain-name northstar.local
crypto key generate rsa modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 2

username admin privilege 15 secret LabAdminSecret123!

! --- Console & VTY Line Protections ---
line con 0
 login local
 exec-timeout 5 0
 logging synchronous
exit

line vty 0 4
 transport input ssh
 login local
 exec-timeout 5 0
 logging synchronous
exit

! --- Centralized DHCP Address Exclusions ---
! Exclude default gateways and static reservation ranges
ip dhcp excluded-address 10.10.10.1 10.10.10.9
ip dhcp excluded-address 10.10.20.1 10.10.20.4
ip dhcp excluded-address 10.10.30.1 10.10.30.9

! --- DHCP Pool: Administration (VLAN 10) ---
ip dhcp pool ADMIN_POOL
 network 10.10.10.0 255.255.255.224
 default-router 10.10.10.1
 dns-server 1.1.1.1
 domain-name northstar.local
exit

! --- DHCP Pool: Finance (VLAN 20) ---
ip dhcp pool FINANCE_POOL
 network 10.10.20.0 255.255.255.240
 default-router 10.10.20.1
 dns-server 1.1.1.1
 domain-name northstar.local
exit

! --- DHCP Pool: IT & Technical Services (VLAN 30) ---
ip dhcp pool IT_POOL
 network 10.10.30.0 255.255.255.224
 default-router 10.10.30.1
 dns-server 1.1.1.1
 domain-name northstar.local
exit

! --- Physical Trunk Interface Enablement ---
interface GigabitEthernet0/0
 description TRUNK_UPLINK_TO_SW1_FA0_1
 no ip address
 no shutdown
exit

! --- Router-on-a-Stick (ROAS) Subinterfaces ---

! Subinterface for VLAN 10 (Administration)
interface GigabitEthernet0/0.10
 description DEFAULT_GATEWAY_VLAN10_ADMIN
 encapsulation dot1Q 10
 ip address 10.10.10.1 255.255.255.224
exit

! Subinterface for VLAN 20 (Finance)
interface GigabitEthernet0/0.20
 description DEFAULT_GATEWAY_VLAN20_FINANCE
 encapsulation dot1Q 20
 ip address 10.10.20.1 255.255.255.240
exit

! Subinterface for VLAN 30 (IT & Infrastructure)
interface GigabitEthernet0/0.30
 description DEFAULT_GATEWAY_VLAN30_IT
 encapsulation dot1Q 30
 ip address 10.10.30.1 255.255.255.224
exit

! Subinterface for VLAN 99 (Management & Native)
interface GigabitEthernet0/0.99
 description DEFAULT_GATEWAY_VLAN99_MGMT_NATIVE
 encapsulation dot1Q 99 native
 ip address 10.10.99.1 255.255.255.240
exit

! --- Simulated WAN / ISP Perimeter Link ---
interface GigabitEthernet0/1
 description SIMULATED_WAN_EDGE_TO_ISP
 ip address 192.0.2.2 255.255.255.252
 no shutdown
exit

! Simulated Default Route to ISP Gateway
ip route 0.0.0.0 0.0.0.0 192.0.2.1

end
write memory
